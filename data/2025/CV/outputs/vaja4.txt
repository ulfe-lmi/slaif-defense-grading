                n = y - k*x;
                if n >= n_min && n <= n_max
                    nj = round((n - n_min) / (n_max - n_min) * (num_n - 1)) + 1;
                    A(ki, nj) = A(ki, nj) + 1;
                end
            end
        end
    end
end

% ------------------------------
% Funkcija drawLines (lokalna)
% ------------------------------
function drawLines(imageRGB, k_list, n_list)
    [H, W, ~] = size(imageRGB);
    figure; imshow(imageRGB); hold on;
    x_vals = 1:W;
    colors = lines(max(1,length(k_list))); % lep nabor barv
    for i = 1:length(k_list)
        k = k_list(i);
        n = n_list(i);
        y_vals = k * x_vals + n;
        valid = y_vals >= 1 & y_vals <= H;
        plot(x_vals(valid), y_vals(valid), 'LineWidth', 2, 'Color', colors(mod(i-1,size(colors,1))+1,:));
        % napiši labelo ob sredini premice
        midx = round(W/2);
        midy = k*midx + n;
        if midy >= 1 && midy <= H
            text(midx, midy, sprintf('%d', i), 'Color', 'y', 'FontSize', 12, 'FontWeight', 'bold');
        end
    end
    title(['Najdenih premic: ', num2str(length(k_list))]);
    hold off;
end

% HoughSpace = HoughTransform(BinaryImage)

k=-2:0.01:2;
n=-200:1:200;
maximum=zeros(N,2);
for p=1:N % (k,n)
[max1,index]=max(HoughSpace);
[max2,columnId]=max(max1);
rowId=index(columnId);
maximum(p,1)=k(rowId);
maximum(p,2)=n(columnId);
HoughSpace(rowId,columnId)=0;
for i=1:481

function [] = drawLines(image, HoughSpace, N, Operator)
    k = -2:0.01:2;
    n = -200:1:200;

    maximum = zeros(N,2);
    [rows, cols, ~] = size(image);

    for p = 1:N % find top N lines
        [max1, index] = max(HoughSpace);      % max of each column
        [max2, columnId] = max(max1);         % global max
        rowId = index(columnId);              % corresponding row
        maximum(p,1) = k(rowId);
        maximum(p,2) = n(columnId);
        HoughSpace(rowId,columnId) = 0;      % remove this peak

        % Draw the line
        x_vals = 1:cols;
        y_vals = round(k(rowId)*x_vals + n(columnId));

        % Keep points inside image
        valid = y_vals >= 1 & y_vals <= rows;

        % Set line color (red)
        for i = 1:length(x_vals)
            if valid(i)
                image(y_vals(i), x_vals(i), 1) = 255;
                image(y_vals(i), x_vals(i), 2) = 0;
                image(y_vals(i), x_vals(i), 3) = 0;
            end
        end
    end

    figure('Name', Operator);
    imshow(image);
end

                        n_idx = (n + 200) * 400/400 + 1;
                        acc(k_idx, n_idx) = acc(k_idx, n_idx) + 1;
                    end
                end
            end
        end
    end

% Author: Marija Ivanovska
% Description: This function draws N lines with maximum number of votes,
% defined by Hough transformation.
% Usage example:
% hotelRGB=imread('hotel.jpg'); % input image
% N = 5; % number of lines we want to draw
% HoughSpace = HoughTransform(BinaryImage)
% drawLines(hotelRGB,HoughSpace,N,'Sobel');
function []=drawLines(image,HoughSpace,N,Operator)
k=-2:0.01:2;
n=-200:1:200;
maximum=zeros(N,2);
for p=1:N % (k,n)
[max1,index]=max(HoughSpace);
[max2,columnId]=max(max1);
rowId=index(columnId);
maximum(p,1)=k(rowId);
maximum(p,2)=n(columnId);
HoughSpace(rowId,columnId)=0;
for i=1:481
    j=round(k(rowId)*i+n(columnId));
    for widthVar=1:size(j)
        image(j,i,1)=255;
        image(j,i,2)=0;
        image(j,i,3)=0;
    end
end
end

function A = HoughTransform(edges)

    [H, W] = size(edges);

    % diskretizacija k in n parametrov
    k_vals = -2:0.01:2;       
    n_vals = -200:1:200;      % 401 values

    num_k = length(k_vals);
    num_n = length(n_vals);

    %  priprava Houghovega prostora
    A = zeros(num_k, num_n);
    
    [y_idxs, x_idxs] = find(edges);

    % zanka čez vse robove
    for idx = 1:length(x_idxs)
        x = x_idxs(idx);
        y = y_idxs(idx);

        for ik = 1:num_k
            k = k_vals(ik);

            n = y - k*x;

            if n < -200 || n > 200
                continue;
            end

            % index for n = round((n - n_min) / step) + 1
            in = round(n - (-200)) + 1;    % step = 1 → easy

            A(ik, in) = A(ik, in) + 1;
        end
    end
end

function [label, color] = doloci_igralca(frame, bb)

    % pragovi
    P1_R_min = 110; P1_R_max = 170;
    P1_G_min = 50;  P1_G_max = 70;
    P1_B_min = 60;  P1_B_max = 110;

    bb = round(bb);

    roi = frame(bb(2):bb(2)+bb(4)-1, ...
                bb(1):bb(1)+bb(3)-1, :);

    R = roi(:,:,1);
    G = roi(:,:,2);
    B = roi(:,:,3);

    % maska rdečega znotraj ROI
    red_mask = (R >= P1_R_min & R <= P1_R_max & ...
                G >= P1_G_min & G <= P1_G_max & ...
                B >= P1_B_min & B <= P1_B_max);

    % delež rdečih pikslov
    red_ratio = sum(red_mask(:)) / numel(red_mask);

    % prag – dovolj že majhen delež
    if red_ratio > 0.03
        label = 'Rdeci igralec';
        color = 'r';
    else
        label = 'Beli igralec';
        color = 'b';
    end
end

%% Funkcije

function [A, kVals, nVals, lines] = houghova_transformacija(BW, Nlines, kMin, kMax, nMin, nMax, numParts, supK, supN)
% Houghova transformacija za premice y = kx + n
% A je velikosti 401x401 (n indeks po vrsticah, k indeks po stolpcih)

    dk = (kMax - kMin)/numParts;
    dn = (nMax - nMin)/numParts;

    kVals = kMin:dk:kMax; % 401
    nVals = nMin:dn:nMax; % 401

    Nk = numel(kVals);
    Nn = numel(nVals);

    [y, x] = find(BW);         % robne točke (x=stolpec, y=vrstica)
    x = double(x);
    y = double(y);


    A = zeros(Nn, Nk, 'uint16');

    % Glasovanje: za vsak k izračunaj n = y - kx, potem akumuliraj
    for ki = 1:Nk
        nAll = y - kVals(ki)*x;                     % n za vse robne točke pri tem k

        nIdx = round((nAll - nMin)/dn) + 1;         % mapiranje na indeks 1..Nn
        valid = (nIdx >= 1) & (nIdx <= Nn);

        % štetje glasov za n indekse
        counts = accumarray(nIdx(valid), 1, [Nn, 1]);
        A(:,ki) = uint16(counts);
    end

    % Poišči Nlines najmočnejših maksimumov
    Awork = double(A);
    lines = struct('k', {}, 'n', {}, 'votes', {}, 'kIdx', {}, 'nIdx', {});

    for t = 1:Nlines
        [maxVal, linIdx] = max(Awork(:));
        if maxVal <= 0
            break;
        end

        [nI, kI] = ind2sub(size(Awork), linIdx);

        lines(t).k = kVals(kI);
        lines(t).n = nVals(nI);
        lines(t).votes = round(maxVal);
        lines(t).kIdx = kI;
        lines(t).nIdx = nI;

        % supresija okolice
        kLo = max(1, kI - supK);  kHi = min(Nk, kI + supK);
        nLo = max(1, nI - supN);  nHi = min(Nn, nI + supN);
        Awork(nLo:nHi, kLo:kHi) = 0;
    end
end

function draw_lines(lines, imgSize)
% Izriši y = kx + n čez sliko (od x=1 do x=W)
    H = imgSize(1);
    W = imgSize(2);

    for i = 1:numel(lines)
        k = lines(i).k;
        n = lines(i).n;

        x1 = 1;   y1 = k*x1 + n;
        x2 = W;   y2 = k*x2 + n;

        plot([x1 x2], [y1 y2], 'r', 'LineWidth', 1);

    end
end

% pomožna funkcija: RGB evklidska razlika ---
function D = rgb_evklidska_razdalja(frameRGB, bgRGB)
    fr = double(frameRGB);
    bg = double(bgRGB);
    dR = fr(:,:,1) - bg(:,:,1);
    dG = fr(:,:,2) - bg(:,:,2);
    dB = fr(:,:,3) - bg(:,:,3);
    D  = sqrt(dR.^2 + dG.^2 + dB.^2);
end

function H = computeHistogramHS(rgbData, nH, nS, satThr, valThr)
% 2D histogram v HSV prostoru (H,S).
% Vrne nH x nS, normalizirano.

    rgbData = im2double(rgbData);
    hsv = rgb2hsv(rgbData);

    Hh = hsv(:,:,1);    % [0,1)
    Ss = hsv(:,:,2);    % [0,1]
    Vv = hsv(:,:,3);    % [0,1]

    mask = (Ss > satThr) & (Vv > valThr);  % izloči “bele/sive” in temne piksle

    h = Hh(mask);
    s = Ss(mask);

    if isempty(h)
        H = zeros(nH, nS);
        return;
    end

    hBin = floor(h * nH) + 1;
    sBin = floor(s * nS) + 1;

    hBin = min(max(hBin, 1), nH);
    sBin = min(max(sBin, 1), nS);

    H = accumarray([hBin, sBin], 1, [nH, nS], @sum, 0);

    H = H / max(1, sum(H(:)));
end

function score = histIntersection2D(H1, H2)
% intersection score (vecji = bolj podobno)
    score = sum(min(H1, H2), 'all');
end

% 2. Naloga

k_min = -2; k_max =  2;
n_min = -200; n_max =  200;

k_vrednosti = linspace(k_min, k_max, 401);  % imamo 400 delov, ampak +1, ker so deli določeni s točkami na robu
n_vrednosti = linspace(n_min, n_max, 401);  % s tema dvema vrsticama razdelimo k in n na točke

dk = k_vrednosti(2) - k_vrednosti(1);
dn = n_vrednosti(2) - n_vrednosti(1);                 % definiramo korak med točkama

H = zeros(length(k_vrednosti),length(n_vrednosti));   % definiramo prazno akumulatorsko matriko 401x401: toliko vrstic, kot je vrednosti k (različnih naklonov) in toliko stolpcev, kot je vrednosti n (različnih odsekov)
                                                      % vsaka celica H(i,j) predstavlja konkretno premico y = k_i * x + n_j
                                                      % vrednost te celice H(i,j) pove, koliko robnih pikslov v sliki verjame, da ležijo na tej premici, 
                                                      % torej akumulator je števec glasov za vsako možno celico

[vrstice, stolpci] = find(C);      % funckija find najde neničelne elemente, vrne v katerih vrsticah in stolpcih so - najde torej vse robne piksle
st_tock = length(vrstice);         % pove koliko robnih pikslov je

for t = 1:st_tock                      % po vseh robnih pikslih
    x = stolpci(t);                    % stolpec je x koordinata piksla
    y = vrstice(t);                    % vrstica je y koordinata piksla
    
    for i_k = 1:length(k_vrednosti)    % po vseh možnih k vrednostih
        k = k_vrednosti(i_k);          % določimo k
        n = y - k * x;                 % določimo n iz enačbe premice

        if n < n_min || n > n_max
            continue;
        end

        j_n = round((n - n_min)/dn) + 1;   % pretvorimo n v indeks j po n_vrednosti

        H(i_k, j_n) = H(i_k, j_n) + 1;     % povečamo števec glasov
    end
end
