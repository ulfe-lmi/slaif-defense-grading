%% Vaja 5 – 2.1: Sledenje igralca (Evklidska razdalja v RGB, brez učenja)

imshow(frameRGB);
title('Z dvema klikoma oznaci igralca (ROI naj bo majhen in informativen)');
[x, y] = ginput(2);

x = round(x); y = round(y);
x1 = min(x); x2 = max(x);
y1 = min(y); y2 = max(y);

w = x2 - x1;
h = y2 - y1;

template = frameRGB(y1:y2, x1:x2, :);   % RGB template

%% Parametri
epsilon = 25;      
numFrames = 150;   

writer = VideoWriter('sledenje_L2_RGB.avi');
open(writer);

prevY = y1;

k = 1;
while hasFrame(v)

    frameRGB = readFrame(v);

    bestDist = inf;
    bestX = prevX;
    bestY = prevY;

    for dx = -epsilon:epsilon
        for dy = -epsilon:epsilon
            x = prevX + dx;
            y = prevY + dy;

            if x < 1 || y < 1 || ...
               x+w > size(frameRGB,2) || ...
               y+h > size(frameRGB,1)
                continue;
            end

            patch = frameRGB(y:y+h, x:x+w, :);

            diff = double(patch) - double(template);
            dist = mean(diff(:).^2);    % L2 v RGB (normaliziran)

            if dist < bestDist
                bestDist = dist;
                bestX = x;
                bestY = y;
            end
        end
    end

    prevX = bestX;
    prevY = bestY;

    imshow(frameRGB); hold on;
    rectangle('Position',[bestX bestY w h], 'EdgeColor','r','LineWidth',2);
    hold off;

    writeVideo(writer, getframe(gca));
    k = k + 1;
end

close(writer);
disp('L2 RGB sledenje koncano');

%% Vaja 5 – 2.3: Sledenje igralca (RGB histogram + Bhattacharyya)
clear; close all; clc;

v = VideoReader('squash.avi');

frameRGB = readFrame(v);

figure(1); imshow(frameRGB);
title('Z dvema klikoma oznaci igralca');
[x, y] = ginput(2);

x = round(x); y = round(y);
x1 = min(x); x2 = max(x);
y1 = min(y); y2 = max(y);

w = x2 - x1;
h = y2 - y1;

template = frameRGB(y1:y2, x1:x2, :);

%% Histogram značilnica (RGB posebej)
numBins = 16;
templateHist = rgbHist(template, numBins);

%% Parametri
epsilon = 15;   
alpha = 0.05;      % update histogram 


writer = VideoWriter('sledenje_hist_bhatt_RGB.avi');
open(writer);

prevX = x1;
prevY = y1;
k = 1;

while hasFrame(v) 
    frameRGB = readFrame(v);

    bestCost = inf;
    bestX = prevX;
    bestY = prevY;

    for dx = -epsilon:epsilon
        for dy = -epsilon:epsilon
            x = prevX + dx;
            y = prevY + dy;

            if x < 1 || y < 1 || ...
               x+w > size(frameRGB,2) || ...
               y+h > size(frameRGB,1)
                continue;
            end

            patch = frameRGB(y:y+h, x:x+w, :);
            patchHist = rgbHist(patch, numBins);

            % Bhattacharyya distance
            bc = sum(sqrt(patchHist .* templateHist));
            dist = 1 - bc;

            % kazen za velike premike 
            motionPenalty = 1e-4 * (dx^2 + dy^2);
            cost = dist + motionPenalty;

            if cost < bestCost
                bestCost = cost;
                bestX = x;
                bestY = y;
            end
        end
    end

    prevX = bestX;
    prevY = bestY;

    % Posodabljanje histograma 
    if alpha > 0
        newPatch = frameRGB(bestY:bestY+h, bestX:bestX+w, :);
        newHist = rgbHist(newPatch, numBins);
        templateHist = (1-alpha)*templateHist + alpha*newHist;
        templateHist = templateHist / sum(templateHist);
    end

    imshow(frameRGB); hold on;
    rectangle('Position',[bestX bestY w h], 'EdgeColor','b','LineWidth',2);
    hold off;

    writeVideo(writer, getframe(gca));
    k = k + 1;
end

close(writer);
disp('Histogram Bhattacharyya (RGB) sledenje koncano');

%% -------- lokalna funkcija --------
function h = rgbHist(imgRGB, numBins)
    imgRGB = uint8(imgRGB);

    r = imgRGB(:,:,1);
    g = imgRGB(:,:,2);
    b = imgRGB(:,:,3);

    hr = imhist(r, numBins);
    hg = imhist(g, numBins);
    hb = imhist(b, numBins);

    h = double([hr; hg; hb]);
    h = h / sum(h);
end

    bestX = prevX;

        for dy = -epsilon:epsilon
            x = prevX + dx;
            y = prevY + dy;

            if x < 1 || y < 1 || ...
               x+w > size(frameRGB,2) || ...
               y+h > size(frameRGB,1)
                continue;
            end

            patch = frameRGB(y:y+h, x:x+w, :);
            patchHist = rgbHist(patch, numBins);

            % Bhattacharyya distance
            bc = sum(sqrt(patchHist .* templateHist));
            dist = 1 - bc;

            % kazen za velike premike- mejhna)
            motionPenalty = 1e-4 * (dx^2 + dy^2);
            cost = dist + motionPenalty;

            if cost < bestCost
                bestCost = cost;
                bestX = x;
                bestY = y;
            end
        end
    end

    prevX = bestX;
    prevY = bestY;

    % Posodabljanje histograma 
    if alpha > 0
        newPatch = frameRGB(bestY:bestY+h, bestX:bestX+w, :);
        newHist = rgbHist(newPatch, numBins);
        templateHist = (1-alpha)*templateHist + alpha*newHist;
        templateHist = templateHist / sum(templateHist);
    end

    imshow(frameRGB); hold on;
    rectangle('Position',[bestX bestY w h], 'EdgeColor','b','LineWidth',2);
    hold off;

    writeVideo(writer, getframe(gca));

function h = rgbHist(imgRGB, numBins)
    imgRGB = uint8(imgRGB);

    r = imgRGB(:,:,1);
    g = imgRGB(:,:,2);
    b = imgRGB(:,:,3);

    hr = imhist(r, numBins);
    hg = imhist(g, numBins);

function H = hist(slika,nBins)
    slika = double(slika) / 255;
    R = slika(:,:,1);
    G = slika(:,:,2);
    B = slika(:,:,3);
    
    r_idx = ceil(R * nBins);
    g_idx = ceil(G * nBins);
    b_idx = ceil(B * nBins);
    
    r_idx(r_idx==0) = 1;
    g_idx(g_idx==0) = 1;
    b_idx(b_idx==0) = 1;
    
    H = zeros(nBins, nBins, nBins);
    for i = 1:numel(R)
        r = r_idx(i);
        g = g_idx(i);
        b = b_idx(i);
        H(r,g,b) = H(r,g,b) + 1;
    end
    H = H/sum(H(:));
end

function razlika = razlika_histogramov_hi_kvadrat(reference, new)
    
    % Hi kvadrat razdalja
    reference = reference(:);
    new = new(:);

    eps_val = 1e-10;    % Prepreči deljenje z nič

% Funkcija naredi histogram podane slike v 4x4x4 bine
function hist = histogram_igralca(image)

    % Če je prazna slika, vrne prazen histogram
    if isempty(image) || size(image,3) ~= 3
    hist = zeros(4,4,4);
    return;
    end

    % Bins
    R = image(:,:,1);
    G = image(:,:,2);
    B = image(:,:,3);
    
    % Pretvorba v vektorje
    R = double(R(:));
    G = double(G(:));
    B = double(B(:));
    
    nBins = 4;
    R_bin = min(floor(R / 256 * nBins), nBins-1);
    G_bin = min(floor(G / 256 * nBins), nBins-1);
    B_bin = min(floor(B / 256 * nBins), nBins-1);
    
    hist_ref = zeros(nBins, nBins, nBins);
    for i = 1:length(R_bin)
        hist_ref(R_bin(i)+1, G_bin(i)+1, B_bin(i)+1) = ...
            hist_ref(R_bin(i)+1, G_bin(i)+1, B_bin(i)+1) + 1;
    end
    
    % Normalize reference histogram to sum to 1 (probability distribution)
    hist = hist_ref / sum(hist_ref(:));
end

function hist = histogram_igralca_sivinski(image)

    % Če je prazna slika, vrne prazen histogram
    if isempty(image)
        hist = zeros(256, 1);
        return;
    end

    % Če je slika RGB, jo pretvori v sivinsko
    if size(image,3) == 3
        % Standardna luminančna pretvorba
        image = 0.2989 * double(image(:,:,1)) + ...
                0.5870 * double(image(:,:,2)) + ...
                0.1140 * double(image(:,:,3));
    else
        image = double(image);
    end

    % Pretvorba v vektor
    gray = image(:);

    % Število binov
    nBins = 8;

    % Izračun binov
    gray_bin = min(floor(gray / 256 * nBins), nBins-1);

    % Histogram
    hist_ref = zeros(nBins, 1);
    for i = 1:length(gray_bin)
        hist_ref(gray_bin(i) + 1) = hist_ref(gray_bin(i) + 1) + 1;
    end

    % Normalizacija (verjetnostna porazdelitev)
    hist = hist_ref / sum(hist_ref);
end

function hist = histogramH(img, locljivost)

    slikaHSV = rgb2hsv(img);
    H = slikaHSV(:,:,1); % H od 0 do 1
    height = size(H,1);
    width = size(H,2);

    hist = zeros(1,locljivost);
    
    for col = 1:width
        for row = 1:height
            bin = floor(H(row,col) * locljivost) + 1; % +1 ker se zacne z 1
            bin = min(max(bin,1), locljivost);
            hist(bin) = hist(bin) + 1;
        end
    end

    maxValue = max(hist(:));
    % vrednost najmocnejsega bina == 1
    hist_norm = (double(hist) / double(maxValue));

    hist = hist_norm;
    
end

function [xBest,yBest] = searchSSD(img, ref, x0, y0, W, H, epsN)

Himg = size(img,1);
Wimg = size(img,2);

xMin = max(1,x0-epsN);
xMax = min(Wimg-W+1,x0+epsN);
yMin = max(1,y0-epsN);
yMax = min(Himg-H+1,y0+epsN);

lambda = 0.05;   % kazen za skoke

bestVal = inf;
xBest = x0; yBest = y0;

for y = yMin:yMax
    for x = xMin:xMax
        p = img(y:y+H-1, x:x+W-1, :);
        d = sum((p(:)-ref(:)).^2);
        penalty = lambda*((x-x0)^2 + (y-y0)^2);
        score = d + penalty;

        if score < bestVal
            bestVal = score;
            xBest = x;
            yBest = y;
        end
    end
end
end

function [xBest,yBest] = searchChi2(imgU8, refH, x0, y0, W, H, epsN, bins)

Wimg = size(imgU8,2);

xMax = min(Wimg-W+1,x0+epsN);
yMin = max(1,y0-epsN);
yMax = min(Himg-H+1,y0+epsN);

bestVal = inf;
xBest = x0; yBest = y0;

for y = yMin:yMax
    for x = xMin:xMax
        patch = imgU8(y:y+H-1, x:x+W-1);
        h = histN(patch,bins);

            xBest = x;
            yBest = y;
        end
    end
end

function h = rgb_hist_fast(img, bins)

img = uint8(img);     %poskrbi da je 0..255

edges = linspace(0,256,bins+1);  %robovi binov (npr. 0-32-...-256)

hR = histcounts(img(:,:,1), edges); %histogram rdecega kanala
hG = histcounts(img(:,:,2), edges); %histogram zelenega kanala
hB = histcounts(img(:,:,3), edges); %histogram modrega kanala

h = double([hR hG hB]);       %zdruzi v en vektor RGB
h = h / sum(h);    %normalizacija
end

function histogram=poisci_histogram(vzorec)
    dim_vzorca=size(vzorec);
    histogram=zeros(8,8,8);
    for R_vrednost=0:32:224
        for G_vrednost=0:32:224
            for B_vrednost=0:32:224
                for j=1:dim_vzorca(2)
                    for i=1:dim_vzorca(1)
                        if ((vzorec(i,j,1)>=R_vrednost) && (vzorec(i,j,1)<=(R_vrednost+31)) && (vzorec(i,j,2)>=G_vrednost) && (vzorec(i,j,2)<=(G_vrednost+31)) && (vzorec(i,j,3)>=B_vrednost) && (vzorec(i,j,3)<=(B_vrednost+31)))
                            histogram((R_vrednost/32)+1,(G_vrednost/32)+1,(B_vrednost/32)+1)=histogram((R_vrednost/32)+1,(G_vrednost/32)+1,(B_vrednost/32)+1)+1;
                        end
                    end
                end
            end
        end
    end
end

function histogram = get_histogram(image, num_bins)
    step = 256/num_bins;
    
    R_idx = min(floor(image(:,:,1) / step) + 1, num_bins);
    G_idx = min(floor(image(:,:,2) / step) + 1, num_bins);
    B_idx = min(floor(image(:,:,3) / step) + 1, num_bins);
    
    rgb_bins = accumarray([R_idx(:), G_idx(:), B_idx(:)], 1, [num_bins num_bins num_bins]);

    % histogram = round(rgb_bins / (max(rgb_bins(:))/255));
    histogram = rgb_bins / sum(rgb_bins(:));
end

%% funkcija histogram sivinske slike
function h = myHistGray(I, nbins, binSize)
    h = zeros(1, nbins);
    pixels = I(:);

    for i = 1:numel(pixels)
        bin = floor(double(pixels(i)) / binSize) + 1;
        if bin > nbins
            bin = nbins;
        end
        h(bin) = h(bin) + 1;
    end

function h = myHistGray(I, nbins, binSize)
    h = zeros(1, nbins);
    pixels = I(:);

    for i = 1:numel(pixels)
        bin = floor(double(pixels(i)) / binSize) + 1;
        if bin > nbins
            bin = nbins;
        end
        h(bin) = h(bin) + 1;
    end
end

function make_resized_video(inFile, outFile, scale, method, antialiasFlag)
    vr = VideoReader(inFile);

    vw = VideoWriter(outFile, 'Motion JPEG AVI');
    vw.FrameRate = vr.FrameRate;
    open(vw);

    while hasFrame(vr)
        frame = readFrame(vr);

        
        if isempty(antialiasFlag)
            frameSmall = imresize(frame, scale, method);
        else
            frameSmall = imresize(frame, scale, method, 'Antialiasing', antialiasFlag);
        end

        writeVideo(vw, frameSmall);
    end

    close(vw);
end

function interpolacija(vhodni_video, izhodni_video, faktor_skaliranja, metoda)
% uporabnik vnese ime vhodne datoteke, ime izhodne datoteke, faktor skaliranja in metodo interpolacije
% imeni vhodne in izhodne datoteke sta tipa string
% faktor skaliranja je številski npr. double
% METODA MORA BITI PODANA NA ENEGA IZMED TEH TREH NAČINOV: 'nearest', 'bilinear_aa_off', 'bilinear_aa_on'


    metoda = string(lower(metoda));             % lower za vsak slučaj naredi vnos neobčutljiv na male in velike črke
    
    video = VideoReader(vhodni_video);

    interpoliran_video = VideoWriter(izhodni_video);
    interpoliran_video.FrameRate = video.FrameRate;
    open(interpoliran_video)

    while hasFrame(video)
        frame = readFrame(video);

        if metoda == 'nearest'
            interpoliran_frame = imresize(frame, faktor_skaliranja, 'nearest');

        elseif metoda == 'bilinear_aa_off'
            interpoliran_frame = imresize(frame, faktor_skaliranja, 'bilinear', 'Antialiasing', false);

        elseif metoda == 'bilinear_aa_on'
            interpoliran_frame = imresize(frame, faktor_skaliranja, 'bilinear', 'Antialiasing', true);
        end
            
        writeVideo(interpoliran_video, interpoliran_frame);
    end

    close(interpoliran_video)
end

figure;

title('Izberite levi zgornji rob pravokotnika okoli rdečega igralca in desni spodnji rob.');

[H, W, ~] = size(prvi_frame);

plot([x2 x2], [y1 y2], 'Color', 'b');  

w = x2 - x1 + 1;               % širina pravokotnika
h = y2 - y1 + 1;               % višina pravokotnika

x = x1;

prvi_siv_frame = im2double(rgb2gray(prvi_frame));
okvir_siv = prvi_siv_frame(y1:y2, x1:x2);
okvir_siv_vektor = okvir_siv(:);

okvir_siv_vektor = okvir_siv_vektor - mean(okvir_siv_vektor);
okvir_siv_vektor = okvir_siv_vektor / (std(okvir_siv_vektor) + 1e-6);

sigma = 0.35;
[X, Y] = meshgrid(linspace(-1,1,w), linspace(-1,1,h));
M = exp(-(X.^2 + Y.^2)/(2*sigma^2));
M = M / sum(M(:));
Mvec = M(:);


novi_video = VideoWriter('squash_pravokotnik_L2_po_ucenju1.avi');
novi_video.FrameRate = video.FrameRate;
open(novi_video)

izhodni_frame = insertShape(prvi_frame, 'rectangle', [x y w h]);
writeVideo(novi_video, izhodni_frame);

alpha = 0.02;   % hitrost učenja (0.01..0.05)
tau   = 5;      % prag zaupanja (manjši je strožje)
eps = 25;
% sledenje po okvirjih
vx = 0; vy = 0;          % hitrost (v px/frame)
beta = 0.7;              % glajenje hitrosti (0.5..0.9)
eps_min = 20;
eps_max = 80;

while hasFrame(video)
    frame_rgb = readFrame(video);
    frame_siv = im2double(rgb2gray(frame_rgb));
   
    x_pred = round(x + vx);
    y_pred = round(y + vy);
    x_pred = max(1, min(W - w + 1, x_pred));
    y_pred = max(1, min(H - h + 1, y_pred));
    % določimo meje lokalne okolice
    x_min = max(1, x_pred - eps);
    y_min = max(1, y_pred - eps);
    x_max = min(W - w + 1, x_pred + eps);
    y_max = min(H - h + 1, y_pred + eps);

    najmanjsa_razlika = inf;
    najboljsi_x = x;
    najboljsi_y = y;
    
    vx_new = najboljsi_x - x;
    vy_new = najboljsi_y - y;
    vx = beta*vx + (1-beta)*vx_new;
    vy = beta*vy + (1-beta)*vy_new;
    
    % sprotno učenje
    bestPatchVec = okvir_siv_vektor;  % tu si shranimo najboljsi izsek (normaliziran vektor)
   
       
    for yy = y_min:y_max
        for xx = x_min:x_max
            izsek_siv = frame_siv(yy:yy+h-1, xx:xx+w-1);
            izsek_siv_vektor = izsek_siv(:);

            izsek_siv_vektor = izsek_siv_vektor - mean(izsek_siv_vektor);
            izsek_siv_vektor = izsek_siv_vektor / (std(izsek_siv_vektor) + 1e-6);

            d = okvir_siv_vektor - izsek_siv_vektor;
            razlika = sum(Mvec .* (d.^2));

            lambda = 0.02;
            pen = lambda * ((xx - x)^2 + (yy - y)^2);
            score = razlika + pen;
            
            if score < najmanjsa_razlika
                najmanjsa_razlika = score;
                najboljsi_x = xx;
                najboljsi_y = yy;
                bestPatchVec = izsek_siv_vektor; % če uporabljaš učenje
            end
        end
     end

     
     % če je ujemanje sumljivo slabo, ne posodabljaj in ne premikaj
     if najmanjsa_razlika > tau
         najboljsi_x = x;
         najboljsi_y = y;
     else
         % če je ujemanje dobro -> posodobi značilnico 
         okvir_siv_vektor = (1-alpha)*okvir_siv_vektor + alpha*bestPatchVec;

         % po posodobitvi spet normaliziraj
         okvir_siv_vektor = okvir_siv_vektor - mean(okvir_siv_vektor);
         okvir_siv_vektor = okvir_siv_vektor / (std(okvir_siv_vektor) + 1e-6);
     end

     x = najboljsi_x;
     y = najboljsi_y;

     izhodni_frame = insertShape(frame_rgb, 'rectangle', [x y w h]);
     writeVideo(novi_video, izhodni_frame);
end
